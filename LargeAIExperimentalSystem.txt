# Minimal Autograd Tensor Engine
# =============================
class Tensor:
    """
    Minimal NumPy-backed tensor with autograd.
    - data: numpy.ndarray (float32)
    - grad: numpy.ndarray or None
    - _prev: set of parent Tensors
    - _backward: function to propagate gradients to parents
    """
    def __init__(self, data, requires_grad: bool=True):
        self.data = np.array(data, dtype=np.float32)
        self.grad = None
        self.requires_grad = requires_grad
        self._prev = set()
        self._backward = lambda: None
        self._op = ''

    def __repr__(self):
        return f"Tensor(shape={self.data.shape}, requires_grad={self.requires_grad}, op={self._op})"

    # basic ops
    def __add__(self, other):
        other = other if isinstance(other, Tensor) else Tensor(other, requires_grad=False)
        out = Tensor(self.data + other.data, requires_grad=self.requires_grad or other.requires_grad)
        out._op = '+'
        out._prev = {self, other}
        def _backward():
            if self.requires_grad:
                self.grad = (self.grad + out.grad) if self.grad is not None else out.grad.copy()
            if other.requires_grad:
                other.grad = (other.grad + out.grad) if other.grad is not None else out.grad.copy()
        out._backward = _backward
        return out

    def __sub__(self, other):
        return self + (-1 * other)

    def __mul__(self, other):
        other = other if isinstance(other, Tensor) else Tensor(other, requires_grad=False)
        out = Tensor(self.data * other.data, requires_grad=self.requires_grad or other.requires_grad)
        out._op = '*'
        out._prev = {self, other}
        def _backward():
            if self.requires_grad:
                grad_self = out.grad * other.data
                self.grad = (self.grad + grad_self) if self.grad is not None else grad_self.copy()
            if other.requires_grad:
                grad_other = out.grad * self.data
                other.grad = (other.grad + grad_other) if other.grad is not None else grad_other.copy()
        out._backward = _backward
        return out

    def __matmul__(self, other):
        out = Tensor(self.data.dot(other.data), requires_grad=self.requires_grad or other.requires_grad)
        out._op = '@'
        out._prev = {self, other}
        def _backward():
            if self.requires_grad:
                grad_self = out.grad.dot(other.data.T)
                self.grad = (self.grad + grad_self) if self.grad is not None else grad_self.copy()
            if other.requires_grad:
                grad_other = self.data.T.dot(out.grad)
                other.grad = (other.grad + grad_other) if other.grad is not None else grad_other.copy()
        out._backward = _backward
        return out

    def sum(self, axis=None, keepdims=False):
        out = Tensor(self.data.sum(axis=axis, keepdims=keepdims), requires_grad=self.requires_grad)
        out._op = 'sum'
        out._prev = {self}
        def _backward():
            grad = out.grad
            if axis is None:
                grad_self = np.ones_like(self.data) * grad
            else:
                shape = list(self.data.shape)
                if not keepdims:
                    # expand grad
                    grad = np.expand_dims(grad, axis)
                grad_self = np.ones_like(self.data) * grad
            self.grad = (self.grad + grad_self) if self.grad is not None else grad_self.copy()
        out._backward = _backward
        return out

    def mean(self, axis=None):
        denom = np.prod(self.data.shape) if axis is None else self.data.shape[axis]
        return self.sum(axis=axis) * (1.0 / denom)

    def T(self):
        out = Tensor(self.data.T.copy(), requires_grad=self.requires_grad)
        out._op = 'T'
        out._prev = {self}
        def _backward():
            self.grad = (self.grad + out.grad.T) if self.grad is not None else out.grad.T.copy()
        out._backward = _backward
        return out

    def reshape(self, *shape):
        out = Tensor(self.data.reshape(*shape), requires_grad=self.requires_grad)
        out._op = 'reshape'
        out._prev = {self}
        def _backward():
            self.grad = (self.grad + out.grad.reshape(self.data.shape)) if self.grad is not None else out.grad.reshape(self.data.shape).copy()
        out._backward = _backward
        return out

    def copy(self):
        return Tensor(self.data.copy(), requires_grad=self.requires_grad)

    def relu(self):
        out_data = np.maximum(0, self.data)
        out = Tensor(out_data, requires_grad=self.requires_grad)
        out._op = 'relu'
        out._prev = {self}
        def _backward():
            grad_self = out.grad * (self.data > 0).astype(np.float32)
            self.grad = (self.grad + grad_self) if self.grad is not None else grad_self.copy()
        out._backward = _backward
        return out

    def tanh(self):
        out_data = np.tanh(self.data)
        out = Tensor(out_data, requires_grad=self.requires_grad)
        out._op = 'tanh'
        out._prev = {self}
        def _backward():
            grad_self = out.grad * (1 - out.data**2)
            self.grad = (self.grad + grad_self) if self.grad is not None else grad_self.copy()
        out._backward = _backward
        return out

    def softmax(self, axis=-1):
        e = np.exp(self.data - np.max(self.data, axis=axis, keepdims=True))
        out_data = e / e.sum(axis=axis, keepdims=True)
        out = Tensor(out_data, requires_grad=self.requires_grad)
        out._op = 'softmax'
        out._prev = {self}
        def _backward():
            # vector-Jacobian product: grad * J
            grad = out.grad
            s = out.data
            grad_self = s * (grad - np.sum(grad * s, axis=axis, keepdims=True))
            self.grad = (self.grad + grad_self) if self.grad is not None else grad_self.copy()
        out._backward = _backward
        return out

    def log(self):
        out = Tensor(np.log(self.data + 1e-12), requires_grad=self.requires_grad)
        out._op = 'log'
        out._prev = {self}
        def _backward():
            grad_self = out.grad / (self.data + 1e-12)
            self.grad = (self.grad + grad_self) if self.grad is not None else grad_self.copy()
        out._backward = _backward
        return out

    def backward(self, grad=None):
        if grad is None:
            grad = np.ones_like(self.data, dtype=np.float32)
        self.grad = grad.astype(np.float32)
        topo = []
        visited = set()
        def build(v):
            if id(v) not in visited:
                visited.add(id(v))
                for p in getattr(v, "_prev", []):
                    build(p)
                topo.append(v)
        build(self)
        for node in reversed(topo):
            node._backward()

def glorot_init(in_dim, out_dim):
    scale = math.sqrt(6.0 / (in_dim + out_dim))
    return np.random.uniform(-scale, scale, size=(in_dim, out_dim)).astype(np.float32)

class LinearLayer:
    def __init__(self, in_dim: int, out_dim: int, bias: bool=True):
        self.W = Tensor(glorot_init(in_dim, out_dim), requires_grad=True)
        self.b = Tensor(np.zeros((out_dim,), dtype=np.float32), requires_grad=True) if bias else None

    def __call__(self, x: Tensor) -> Tensor:
        # x: Tensor of shape (..., in_dim)
        orig_shape = x.data.shape
        flat = x.reshape(-1, orig_shape[-1])
        out = flat @ self.W
        if self.b is not None:
            out = out + Tensor(self.b.data.reshape(1, -1), requires_grad=False)
        out = out.reshape(*orig_shape[:-1], out.data.shape[-1])
        return out

    def params(self):
        return [self.W] + ([self.b] if self.b is not None else [])

class LayerNorm:
    def __init__(self, dim: int, eps: float=1e-5):
        self.eps = eps
        self.gamma = Tensor(np.ones((dim,), dtype=np.float32), requires_grad=True)
        self.beta = Tensor(np.zeros((dim,), dtype=np.float32), requires_grad=True)

    def __call__(self, x: Tensor) -> Tensor:
        data = x.data
        mean = data.mean(axis=-1, keepdims=True)
        var = ((data - mean)**2).mean(axis=-1, keepdims=True)
        norm = (data - mean) / np.sqrt(var + self.eps)
        out = norm * self.gamma.data.reshape((1,)*(norm.ndim-1)+(-1,)) + self.beta.data.reshape((1,)*(norm.ndim-1)+(-1,))
        return Tensor(out, requires_grad=False)  # non-differentiable wrapper for simplicity

# Simple Conv1D (naive sliding)
class Conv1D:
    def __init__(self, in_ch: int, out_ch: int, kernel: int, stride: int=1, bias=True):
        self.in_ch = in_ch; self.out_ch = out_ch; self.kernel = kernel; self.stride = stride
        self.W = Tensor(np.random.randn(out_ch, in_ch, kernel).astype(np.float32) * (1.0/math.sqrt(in_ch*kernel)), requires_grad=True)
        self.b = Tensor(np.zeros((out_ch,), dtype=np.float32), requires_grad=True) if bias else None

    def __call__(self, x: Tensor) -> Tensor:
        # x: Tensor shape (B, L, C)
        B,L,C = x.data.shape
        out_len = (L - self.kernel)//self.stride + 1
        out = np.zeros((B, out_len, self.out_ch), dtype=np.float32)
        for i in range(out_len):
            segment = x.data[:, i*self.stride:i*self.stride+self.kernel, :]  # (B, kernel, C)
            # conv: for each out_ch, dot with W[out_ch] of shape (in_ch,kernel)
            for oc in range(self.out_ch):
                w = self.W.data[oc]  # (in_ch,kernel)
                # transpose segment to (B, in_ch, kernel)
                seg_t = segment.transpose(0,2,1)
                out[:, i, oc] = np.tensordot(seg_t, w, axes=([1,2],[0,1]))
        if self.b is not None:
            out = out + self.b.data.reshape((1,1,-1))
        return Tensor(out, requires_grad=False)  # keep simple

def causal_mask(T):
    m = np.triu(np.ones((T,T), dtype=bool), 1)
    return m

class MultiHeadSelfAttention:
    def __init__(self, dim: int, heads: int=4):
        assert dim % heads == 0
        self.dim = dim; self.heads = heads; self.dh = dim // heads
        # projection matrices
        self.Wq = Tensor(glorot_init(dim, dim), requires_grad=True)
        self.Wk = Tensor(glorot_init(dim, dim), requires_grad=True)
        self.Wv = Tensor(glorot_init(dim, dim), requires_grad=True)
        self.Wo = Tensor(glorot_init(dim, dim), requires_grad=True)

    def split_heads(self, x_np):
        B,T,D = x_np.shape
        x = x_np.reshape(B, T, self.heads, self.dh).transpose(0,2,1,3)  # (B, heads, T, dh)
        return x

    def combine_heads(self, x):
        B,h,T,dh = x.shape
        return x.transpose(0,2,1,3).reshape(B, T, h*dh)

    def __call__(self, x: Tensor, mask: np.ndarray=None) -> Tensor:
        # x: Tensor(B,T,D)
        X = x.data
        Q = X.dot(self.Wq.data)
        K = X.dot(self.Wk.data)
        V = X.dot(self.Wv.data)
        Qh = self.split_heads(Q)  # (B,h,T,dh)
        Kh = self.split_heads(K)
        Vh = self.split_heads(V)
        # scaled dot-product per head
        scores = np.matmul(Qh, Kh.transpose(0,1,3,2)) / math.sqrt(self.dh)  # (B,h,T,T)
        if mask is not None:
            scores = np.where(mask[None,None,:,:], -1e9, scores)
        probs = np.exp(scores - np.max(scores, axis=-1, keepdims=True))
        probs = probs / probs.sum(axis=-1, keepdims=True)
        out = np.matmul(probs, Vh)  # (B,h,T,dh)
        combined = self.combine_heads(out)  # (B,T,D)
        outT = Tensor(combined.dot(self.Wo.data), requires_grad=False)
        return outT

    def params(self):
        return [self.Wq, self.Wk, self.Wv, self.Wo]

class TransformerBlock:
    def __init__(self, dim: int, heads: int=4, mlp_hidden: int=None):
        self.attn = MultiHeadSelfAttention(dim, heads=heads)
        self.ln1 = LayerNorm(dim)
        self.ln2 = LayerNorm(dim)
        self.mlp = LinearLayer(dim, mlp_hidden or (dim*4))
        self.mlp_out = LinearLayer(mlp_hidden or (dim*4), dim)

    def __call__(self, x: Tensor, mask: np.ndarray=None) -> Tensor:
        # x: Tensor(B,T,D)
        a = self.attn(x, mask=mask)
        x1 = Tensor(x.data + a.data, requires_grad=False)
        x2 = self.ln1(x1)
        f = self.mlp(Tensor(x2.data, requires_grad=False)).tanh()
        f2 = self.mlp_out(f)
        x3 = Tensor(x2.data + f2.data, requires_grad=False)
        x4 = self.ln2(x3)
        return x4

    def params(self):
        p = []
        p += self.attn.params()
        p += self.mlp.params() + self.mlp_out.params()
        p += [self.ln1.gamma, self.ln1.beta, self.ln2.gamma, self.ln2.beta]
        return p

class Expert:
    def __init__(self, dim:int):
        self.lin = LinearLayer(dim, dim)

    def __call__(self, x: Tensor) -> Tensor:
        return self.lin(x).tanh()

    def params(self):
        return self.lin.params()

class MoE:
    def __init__(self, dim:int, n_experts:int=4, k_top:int=2):
        self.experts = [Expert(dim) for _ in range(n_experts)]
        self.router = LinearLayer(dim, n_experts)
        self.k = k_top

    def __call__(self, x: Tensor) -> Tensor:
        # x: Tensor(B,T,D)
        # naive routing: compute softmax over experts per token and weighted-sum
        logits = self.router(x)  # Tensor(B,T,n_experts)
        probs = logits.softmax(axis=-1)  # Tensor with data
        out_data = np.zeros((x.data.shape[0], x.data.shape[1], x.data.shape[2]), dtype=np.float32)
        for i, e in enumerate(self.experts):
            out_data += probs.data[..., i:i+1] * e(x).data
        return Tensor(out_data, requires_grad=False)

    def params(self):
        p = []
        for e in self.experts:
            p += e.params()
        p += self.router.params()
        return p

class VAEEncoder:
    def __init__(self, input_dim:int, latent_dim:int):
        self.fc1 = LinearLayer(input_dim, input_dim*2)
        self.fc_mu = LinearLayer(input_dim*2, latent_dim)
        self.fc_logvar = LinearLayer(input_dim*2, latent_dim)

    def __call__(self, x: Tensor) -> Tuple[Tensor, Tensor]:
        h = self.fc1(x).tanh()
        mu = self.fc_mu(h)
        logvar = self.fc_logvar(h)
        return mu, logvar

class VAEDecoder:
    def __init__(self, latent_dim:int, output_dim:int):
        self.fc1 = LinearLayer(latent_dim, latent_dim*2)
        self.fc_out = LinearLayer(latent_dim*2, output_dim)

    def __call__(self, z: Tensor) -> Tensor:
        h = self.fc1(z).tanh()
        return self.fc_out(h)

# KLD and reconstruction
def kld_loss(mu: Tensor, logvar: Tensor) -> Tensor:
    # KL divergence between N(mu, sigma) and N(0,1)
    var = np.exp(logvar.data)
    kld = 0.5 * (mu.data**2 + var - 1 - logvar.data)
    return Tensor(np.mean(kld), requires_grad=False)

def recon_loss(recon: Tensor, target: Tensor) -> Tensor:
    # L2 loss
    diff = Tensor(recon.data - target.data, requires_grad=False)
    return Tensor(np.mean(diff.data**2), requires_grad=False)

class SimpleDDPMScheduler:
    def __init__(self, timesteps:int=1000, beta_start:float=1e-4, beta_end:float=0.02):
        self.timesteps = timesteps
        self.betas = np.linspace(beta_start, beta_end, timesteps, dtype=np.float32)
        self.alphas = 1.0 - self.betas
        self.alpha_cumprod = np.cumprod(self.alphas, axis=0)

    def q_sample(self, x0: np.ndarray, t:int, noise: np.ndarray):
        # x_t = sqrt(alpha_cumprod[t]) * x0 + sqrt(1-alpha_cumprod[t]) * noise
        ac = self.alpha_cumprod[t]
        return np.sqrt(ac) * x0 + np.sqrt(1.0 - ac) * noise

# Very small UNet-ish denoiser (toy)
class TinyDenoiser:
    def __init__(self, dim:int):
        self.enc = LinearLayer(dim, dim)
        self.mid = LinearLayer(dim, dim)
        self.dec = LinearLayer(dim, dim)

    def __call__(self, x: np.ndarray) -> np.ndarray:
        # x: numpy array (B, D)
        h = np.tanh(self.enc(Tensor(x, requires_grad=False)).data)
        h2 = np.tanh(self.mid(Tensor(h, requires_grad=False)).data)
        out = self.dec(Tensor(h2, requires_grad=False)).data
        return out

class ToyEnv:
    """
    Simple deterministic environment:
    - state is int in [0, N-1]
    - action {0,1}
    - transition: state = (state + (action*2-1)) mod N
    - reward: +1 if state == goal else 0
    """
    def __init__(self, N=8, goal=0):
        self.N = N; self.goal = goal
        self.state = 0

    def reset(self):
        self.state = random.randrange(self.N)
        return self.state

    def step(self, action:int):
        self.state = (self.state + (action*2-1)) % self.N
        reward = 1.0 if self.state == self.goal else 0.0
        done = False
        return self.state, reward, done, {}

class PolicyNet:
    def __init__(self, obs_dim:int, hidden:int=32):
        self.linear = LinearLayer(obs_dim, hidden)
        self.out = LinearLayer(hidden, 2)  # two actions

    def act(self, obs: np.ndarray) -> int:
        x = np.tanh(self.linear(Tensor(obs.reshape(1,-1), requires_grad=False)).data)
        logits = self.out(Tensor(x, requires_grad=False)).data
        probs = np.exp(logits - np.max(logits))
        probs = probs / probs.sum()
        return int(np.random.choice(len(probs.flatten()), p=probs.flatten()))

# Neural Architecture Search (evolutionary skeleton)
# =============================
class NASPopulation:
    def __init__(self, pop_size:int=10, gene_len:int=32):
        self.pop_size = pop_size
        self.gene_len = gene_len
        self.pop = [np.random.randint(0, 3, size=(gene_len,)) for _ in range(pop_size)]
        self.fitness = [None]*pop_size

    def mutate(self, gene):
        g = gene.copy()
        idx = random.randrange(len(g))
        g[idx] = (g[idx] + random.choice([1,2])) % 3
        return g

    def crossover(self, a, b):
        pt = random.randrange(1, len(a))
        child = np.concatenate([a[:pt], b[pt:]])
        return child

    def step(self, evaluator_fn):
        # evaluate
        for i,g in enumerate(self.pop):
            if self.fitness[i] is None:
                self.fitness[i] = evaluator_fn(g)
        # select top half
        order = np.argsort(self.fitness)[::-1]
        survivors = [self.pop[i] for i in order[:self.pop_size//2]]
        newpop = survivors.copy()
        while len(newpop) < self.pop_size:
            a,b = random.sample(survivors,2)
            c = self.crossover(a,b)
            if random.random() < 0.3:
                c = self.mutate(c)
            newpop.append(c)
        self.pop = newpop
        self.fitness = [None]*self.pop_size

# Learned Optimizer (tiny LSTM skeleton)
# =============================
class TinyLSTMOpt:
    def __init__(self, param_size:int, hidden:int=64):
        # parameters for LSTM cell (very tiny and naive)
        self.W = np.random.randn(param_size+hidden, 4*hidden).astype(np.float32) * 0.02
        self.b = np.zeros((4*hidden,), dtype=np.float32)
        self.hidden = np.zeros((hidden,), dtype=np.float32)
        self.cell = np.zeros((hidden,), dtype=np.float32)
        self.param_size = param_size
        self.hidden_dim = hidden

    def step(self, grads: np.ndarray, params: np.ndarray):
        # naive: treat grads as input vector, run one LSTM step to compute updates
        x = np.concatenate([grads.flatten(), self.hidden], axis=0)
        gates = x @ self.W + self.b  # (4*hidden,)
        i = sigmoid(gates[:self.hidden_dim])
        f = sigmoid(gates[self.hidden_dim:2*self.hidden_dim])
        o = sigmoid(gates[2*self.hidden_dim:3*self.hidden_dim])
        g = np.tanh(gates[3*self.hidden_dim:])
        self.cell = f * self.cell + i * g
        self.hidden = o * np.tanh(self.cell)
        # produce param delta from hidden (project)
        delta = -0.001 * np.tanh(self.hidden[:params.size]).reshape(params.shape)
        params += delta
        return params

def sigmoid(x): return 1.0 / (1.0 + np.exp(-x))

# Replay Memory
# =============================
class ReplayMemory:
    def __init__(self, capacity:int=10000):
        self.capacity = capacity
        self.buffer = []
    def push(self, item):
        if len(self.buffer) >= self.capacity:
            self.buffer.pop(0)
        self.buffer.append(item)
    def sample(self, k:int):
        return random.sample(self.buffer, min(k, len(self.buffer)))

# Synthetic Dataset and Dataloader
# =============================
def synthetic_sequence_dataset(vocab:int=128, seq_len:int=32, size:int=1000):
    X = np.random.randint(0, vocab, size=(size, seq_len))
    return X

def dataloader(X: np.ndarray, batch_size:int=32, shuffle:bool=True):
    idx = np.arange(X.shape[0])
    if shuffle:
        np.random.shuffle(idx)
    for i in range(0, X.shape[0], batch_size):
        batch_idx = idx[i:i+batch_size]
        yield X[batch_idx]

        return pickle.load(f)

# Simple RMSProp optimizer (numpy arrays)
class RMSProp:
    def __init__(self, params: List[np.ndarray], lr=1e-3, alpha=0.99, eps=1e-8):
        self.params = params
        self.lr = lr; self.alpha = alpha; self.eps = eps
        self.sq = [np.zeros_like(p) for p in params]

    def step(self, grads: List[np.ndarray]):
        for i,p in enumerate(self.params):
            self.sq[i] = self.alpha * self.sq[i] + (1-self.alpha) * (grads[i]**2)
            p -= self.lr * grads[i] / (np.sqrt(self.sq[i]) + self.eps)

# High-level "OmegaNet" combining many components
# =============================
class OmegaNet:
    def __init__(self, config: Dict[str,Any]):
        self.dim = config.get('dim', 64)
        self.vocab = config.get('vocab', 256)
        self.depth = config.get('depth', 3)
        self.heads = config.get('heads', 4)
        self.seq_len = config.get('seq_len', 32)
        # embeddings
        self.token_emb = np.random.randn(self.vocab, self.dim).astype(np.float32) * 0.01
        self.pos_emb = np.random.randn(self.seq_len, self.dim).astype(np.float32) * 0.01
        # transformer blocks
        self.blocks = [TransformerBlock(self.dim, heads=self.heads) for _ in range(self.depth)]
        self.moe = MoE(self.dim, n_experts=4)
        self.vocab_head = LinearLayer(self.dim, self.vocab)
        # VAE (for reconstruction tasks)
        self.vae_enc = VAEEncoder(self.seq_len*self.dim, 64)
        self.vae_dec = VAEDecoder(64, self.seq_len*self.dim)

    def forward_logits(self, idx_batch: np.ndarray) -> np.ndarray:
        # idx_batch: (B, T)
        B,T = idx_batch.shape
        emb = self.token_emb[idx_batch] + self.pos_emb[None,:T,:]
        x = Tensor(emb, requires_grad=False)
        mask = causal_mask(T)
        for blk in self.blocks:
            x = blk(x, mask=mask)
        # pass through MoE
        moe_out = self.moe(x)  # Tensor
        logits = self.vocab_head(Tensor(moe_out.data, requires_grad=False)).data
        return logits  # (B, T, V)

# Simple training loops (toy)
# =============================
def train_language_model(config: Dict[str,Any]):
    print("[Train] Starting LM training (toy)...")
    seq_len = config.get('seq_len', 32)
    vocab = config.get('vocab', 128)
    dim = config.get('dim', 64)
    epochs = config.get('epochs', 3)
    batch_size = config.get('batch_size', 16)
    model = OmegaNet(config)
    # dummy params: collect as numpy arrays (for RMSProp)
    params = [model.token_emb, model.pos_emb]
    # also include vocab_head weights
    params += [model.vocab_head.W.data, model.vocab_head.b.data]
    opt = RMSProp(params, lr=1e-3)
    X = synthetic_sequence_dataset(vocab=vocab, seq_len=seq_len, size=512)
    for epoch in range(epochs):
        total_loss = 0.0
        steps = 0
        for batch in dataloader(X, batch_size=batch_size):
            logits = model.forward_logits(batch)  # (B,T,V)
            # compute cross-entropy next-token
            targets = np.roll(batch, -1, axis=1)
            # softmax + nll
            logits_max = np.max(logits, axis=-1, keepdims=True)
            exps = np.exp(logits - logits_max)
            probs = exps / np.sum(exps, axis=-1, keepdims=True)
            # gather out
            B,T,V = probs.shape
            idx = np.arange(B)[:,None], np.arange(T)[None,:], targets
            selected = probs[idx]
            nll = -np.log(selected + 1e-12).mean()
            total_loss += nll
            steps += 1
            # naive gradient step: compute grad wrt logits approximatively and propagate to token_emb and pos_emb via finite differences is expensive;
            # For this toy trainer, perform a crude parameter update on token_emb via policy-gradient style nudging:
            # push embeddings for correct tokens up slightly towards logits
            for b in range(B):
                for t in range(T):
                    gt = targets[b,t]
                    model.token_emb[gt] += 1e-4 * (probs[b,t] - (np.arange(V)==gt))
            if steps % 20 == 0:
                pass
        print(f"Epoch {epoch+1}/{epochs} - Loss {total_loss/steps:.4f}")

# CLI and modes
# =============================
def main():
    parser = argparse.ArgumentParser(description="OMEGA-SUPREME: massive single-file AI playground")
    parser.add_argument('--mode', type=str, default='train', choices=['train','sample','nas','debug'], help='Mode to run')
    parser.add_argument('--config', type=str, default=None, help='Path to JSON config file')
    args = parser.parse_args()
    config = {
        'dim': 64,
        'vocab': 128,
        'depth': 3,
        'heads': 4,
        'seq_len': 32,
        'epochs': 3,
        'batch_size': 16
    }
    if args.config and os.path.exists(args.config):
        with open(args.config,'r') as f:
            userc = json.load(f)
            config.update(userc)
    if args.mode == 'train':
        train_language_model(config)
    elif args.mode == 'sample':
        print("[Sample] Generating toy samples...")
        model = OmegaNet(config)
        # random seed sequence
        idx = np.random.randint(0, config['vocab'], size=(1, config['seq_len']))
        logits = model.forward_logits(idx)
        next_probs = np.exp(logits[0,-1] - logits[0,-1].max())
        next_probs = next_probs / next_probs.sum()
        token = np.random.choice(len(next_probs), p=next_probs)
        print("Next token sampled:", token)
    elif args.mode == 'nas':
        print("[NAS] Running tiny evolutionary NAS...")
        nas = NASPopulation(pop_size=8, gene_len=16)
        def dummy_eval(gene):
            # pretend the gene maps to architectures; score randomly but reproducibly
            return float(np.sum(gene) + random.random())
        for gen in range(8):
            nas.step(dummy_eval)
            print(f"NAS gen {gen+1} top fitness: {max(nas.fitness) if nas.fitness else 'N/A'}")
    elif args.mode == 'debug':
        print("[Debug] Running internal quick checks...")
        t = Tensor(np.array([[1.0,2.0],[3.0,4.0]]))
        s = t.sum()
        s.backward()
        print("Tensor grad test:", t.grad)
    else:
        print("Mode not recognized.")

if __name__ == '__main__':
    main()
